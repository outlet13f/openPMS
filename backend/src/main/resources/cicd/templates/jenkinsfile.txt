// openPMS에 빌드·배포 결과를 알린다. Jenkins 자격 증명(Secret text) 'openpms-webhook-secret'에 <웹훅 비밀>을 넣어 두세요.
// repository는 openPMS 프로젝트에 붙은 저장소 경로(group/name)와 같아야 합니다.
def notifyOpenPms(Map n) {
  withCredentials([string(credentialsId: 'openpms-webhook-secret', variable: 'OPENPMS_SECRET')]) {
    writeFile file: 'openpms-notice.json', text: groovy.json.JsonOutput.toJson(n + [at: new Date().format("yyyy-MM-dd'T'HH:mm:ssXXX")])
    sh '''
      sig=$(openssl dgst -sha256 -hmac "$OPENPMS_SECRET" openpms-notice.json | sed 's/^.* //')
      curl -fsS -X POST -H 'Content-Type: application/json' -H "X-OpenPMS-Signature: sha256=$sig" \
        --data-binary @openpms-notice.json '{{WEBHOOK_URL}}'
    '''
  }
}

//   sha: env.GIT_COMMIT, ref: env.BRANCH_NAME ?: env.GIT_BRANCH, status: currentBuild.currentResult == 'SUCCESS' ? 'success' : 'failed', url: env.BUILD_URL)
